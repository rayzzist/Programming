        string input = Console.ReadLine();

        List<string> textN = input.Split(' ').ToList();

        List<int> numbers = new List<int>();

        foreach (string item in textN)
        {
            int num = int.Parse(item);
            numbers.Add(num);
        }

        foreach (int num in numbers)
        {
            Console.WriteLine(num);
        }
        Console.WriteLine($"броят на числата е: {numbers.Count}");

        List<int> filterednums = new List<int>();

        foreach (int num in numbers)
        {
            if (num > 0 && num % 2 == 0 && !filterednums.Contains(num))
            {
                filterednums.Add(num);
            }
        }

        Console.WriteLine(string.Join(", ", filterednums));

        int count = 0;
        int sum = 0;
        foreach (int n in filterednums)
        {
            count++;
            sum += n;
        }

        Console.WriteLine($"броят на числата: {count}");
        Console.WriteLine($"сумата на числата: {sum}");
    }
